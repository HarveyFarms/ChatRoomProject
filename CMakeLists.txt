cmake_minimum_required(VERSION 3.16)
project(ChatRoomProject LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.2 REQUIRED COMPONENTS Widgets Network)
add_executable(chat_server src/server.cpp)
target_link_libraries(chat_server PRIVATE Qt6::Network)
add_executable(chat_client WIN32 src/client.cpp)
target_link_libraries(chat_client PRIVATE Qt6::Widgets Qt6::Network)
